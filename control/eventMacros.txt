#################################################### FAIXA 20-40 ####################################################
# Nesta faixa o personagem irá para Geffen e depois para gef_fild05
######################################################################################################################

automacro vaiNaCreamy {
    BaseLevel 20..40
    InSaveMap 0
    InLockMap 0
    run-once 1
    timeout 10
    call irParaGeffen
}
automacro creamy {
    BaseLevel 20..40
    run-once 1
    timeout 10
    call configurarGeffenCreamy
}
macro configurarGeffenCreamy {
    # Configurações de combate
        do conf lockMap gef_fild05
        do conf route_randomWalk 1
        do conf route_randomWalk_inTown 0
        do conf attackAuto 2
        do conf attackAuto_inLockOnly 1

        # Configurações de teleporte
        do conf teleportAuto_hp 30
        do conf teleportAuto_sp 50
        do conf teleportAuto_idle 0
        do conf teleportAuto_minAggressivesInLock 3

        # Configurações de itens
        do conf itemsTakeAuto 1
        do conf itemsTakeAuto_party 0

        # Configurações de venda
        do conf sellAuto 1
        do conf sellAuto_npc geffen 193 152
        do conf sellAuto_standpoint
        do conf sellAuto_distance 5
        do conf sellAuto_npc_steps s

    	stop
}

######################################################################################################################
#################################################### SAVE POINT ####################################################
######################################################################################################################

macro irParaGeffen {
	if ($.map ne 'geffen') {
		do move geffen
		pause 3
		do talknpc 120 62 ro
		pause 3
		do conf saveMap $.map
		pause 3
	}
	stop
}
macro irParaAldebaran {
	if ($.map ne 'aldebaran') {
		do move aldebaran
		pause 3
		do talknpc 143 119 ro
		pause 3
		do conf saveMap $.map
		pause 3
	}
	stop
}
